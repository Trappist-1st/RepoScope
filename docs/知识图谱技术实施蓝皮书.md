# Codebase-Memory 技术实施蓝皮书
> 基于论文《Codebase-Memory: Tree-Sitter-Based Knowledge Graphs for LLM Code Exploration via MCP》(Vogel et al., 2026) 的架构提炼，面向 RepoScope 项目的落地评估

---

## 1. 核心方法提炼

论文要解决的核心问题是：LLM 编码代理通过反复的文件读取和 grep 搜索来理解代码库，这种文本式探索无法捕捉调用图、依赖链等结构化关系，导致 token 消耗随代码库规模线性增长。作者提出 Codebase-Memory，用 Tree-Sitter 解析 66 种语言的源码，构建一个存储在单文件 SQLite 中的属性图（property graph），并通过 MCP 协议暴露 14 个结构化查询工具，使 LLM 代理可以直接做图查询而非反复读文件。核心创新点有三个：(1) 六阶段流水线 + 多线程 worker pool 的并行图构建；(2) 一套六策略级联的调用消解算法（import map → 同模块 → 唯一名 → 后缀匹配 → 模糊匹配），并针对 Go/C/C++ 额外叠加 LSP 式类型解析以处理方法接收者和指针间接引用；(3) Louvain 社区发现用于架构层面的模块聚类。评测显示：相较于纯文件探索代理，该方法以 1/10 的 token 消耗、2.1 倍更少的工具调用，达到 90% 的探索代理答案质量，在需要跟踪跨文件关系的查询（枢纽检测、调用者排序）上甚至反超。

## 2. 算法流程伪代码

```python
# ============================================================
# 输入: repo_root (源码目录)
# 输出: KnowledgeGraph(nodes, edges) 持久化于 SQLite
# ============================================================

def build_knowledge_graph(repo_root: str) -> KnowledgeGraph:
    gbuf = GraphBuffer()  # 内存中的临时图缓冲区，按 qualified_name / label / id 建哈希索引

    # ---------- Phase 1: Structure ----------
    # 遍历目录树，生成 Project/Package/Folder/File 节点及 CONTAINS_* 边
    file_list = discover_files(repo_root)
    for f in file_list:
        gbuf.add_node(make_file_node(f))
        gbuf.add_edge(contains_edge(parent_dir(f), f))

    # ---------- Phase 2: Extraction (并行) ----------
    # 每个 worker 处理一个文件子集，写入各自的 buffer，最后 merge
    worker_pool = ThreadPool(n_workers)
    per_worker_bufs = worker_pool.map(extract_definitions, file_list)
    # extract_definitions(file):
    #   ast = tree_sitter_parse(file)
    #   for node in walk(ast):
    #       if is_definition(node):  # 函数/方法/类/接口/枚举/类型
    #           entity = Entity(
    #               name=node.name, qualified_name=qualify(node, file),
    #               kind=node.kind, signature=extract_signature(node),
    #               decorators=node.decorators, is_exported=node.is_exported,
    #               complexity=estimate_complexity(node),
    #           )
    #           buf.add_node(entity)
    #           registry.index_by_qualified_name(entity)
    #           registry.index_by_simple_name(entity)   # 反向索引，供消歧使用
    gbuf = merge_buffers(per_worker_bufs)

    # ---------- Phase 3: Resolution (并行, 核心消歧逻辑) ----------
    call_edges = []
    for f in file_list:
        import_map = parse_imports(f)                  # {alias: module_qualified_name}
        for call_site in find_call_sites(f):
            callee_ref = resolve_callee(call_site, import_map, registry, current_file=f)
            if callee_ref is not None:
                call_edges.append(Edge(
                    src=call_site.caller_qname, dst=callee_ref.qname,
                    type="CALLS", confidence=callee_ref.confidence,
                ))
    gbuf.add_edges(call_edges)
    # 同阶段还产出 IMPORTS / USAGES / USES_TYPE / IMPLEMENTS / INHERITS / DECORATES 边

    # ---------- 消歧核心: 六策略级联 ----------
    def resolve_callee(call_site, import_map, registry, current_file) -> Ref | None:
        prefix, suffix = split_callee(call_site.raw_name)  # e.g. "pkg.Func" -> ("pkg","Func")

        # 策略1: import map 精确匹配 (置信度 0.95)
        if prefix in import_map:
            module_qname = import_map[prefix]
            candidate = registry.exact_lookup(f"{module_qname}.{suffix}")
            if candidate: return Ref(candidate, confidence=0.95)

        # 策略2: import map 后缀匹配 (0.85) —— 精确匹配失败时的回退
        for module_qname in import_map.values():
            candidate = registry.suffix_lookup(module_qname, suffix)
            if candidate: return Ref(candidate, confidence=0.85)

        # 策略3: 同模块前缀匹配 (0.90)
        same_module_qname = f"{module_of(current_file)}.{call_site.raw_name}"
        candidate = registry.exact_lookup(same_module_qname)
        if candidate: return Ref(candidate, confidence=0.90)

        # 策略4: 全局唯一简单名 (0.75)，若 import 不可达则降权
        candidates = registry.reverse_lookup(call_site.raw_name)
        if len(candidates) == 1:
            conf = 0.75 if is_import_reachable(candidates[0], current_file) else 0.60
            return Ref(candidates[0], confidence=conf)

        # 策略5: 多候选按 import 距离评分的后缀匹配 (0.55)
        if len(candidates) > 1:
            best = min(candidates, key=lambda c: import_distance(c, current_file))
            return Ref(best, confidence=0.55)

        # 策略6: 字符串相似度兜底 (0.30~0.40)
        fuzzy = registry.fuzzy_match(call_site.raw_name, threshold=0.30)
        if fuzzy: return Ref(fuzzy, confidence=fuzzy.score)

        return None  # 未消歧，丢弃该调用边或标记为 unresolved

    # ---------- (Go/C/C++ 专用) LSP 式类型解析，绕过字符串级联 ----------
    if lang in ("go", "c", "cpp"):
        type_registry = build_type_registry(all_files, stdlib_stubs)
        for f in files_of(lang):
            scope = Scope()
            track_bindings(f, scope)  # 变量声明/赋值/短声明/形参
            for call_site in find_call_sites(f):
                recv_type = eval_receiver_type_bottom_up(call_site, scope, type_registry)
                # 字段/方法查找 -> 基类/内嵌类型遍历 -> 返回类型沿调用链传播 -> 类型化简
                resolved = type_registry.lookup_method(recv_type, call_site.method_name)
                if resolved:
                    call_edges.append(Edge(call_site.caller_qname, resolved.qname,
                                            type="CALLS", confidence=1.0, method="type_resolved"))

    # ---------- Phase 4: Enrichment ----------
    add_test_edges(gbuf)                # TESTS
    add_http_route_edges(gbuf, extractors=FRAMEWORK_EXTRACTORS)  # HTTP_CALLS / HANDLES
    add_config_edges(gbuf)              # CONFIGURES
    add_git_cochange_edges(gbuf)        # FILE_CHANGES_WITH

    # ---------- Phase 5: Flush ----------
    with sqlite_transaction() as tx:
        remap_temp_ids_to_row_ids(gbuf)
        bulk_insert(tx, gbuf.nodes, gbuf.edges)
        create_indexes(tx)  # 延迟建索引，减少写入开销

    # ---------- Phase 6: Post-index ----------
    communities = louvain_communities(gbuf.call_graph_subview())  # 见第3节公式
    persist_communities(communities)
    persist_content_hashes(file_list, algo="XXH3")

    return KnowledgeGraph(store="sqlite", tools=MCP_TOOL_SET)


def incremental_reindex(changed_file):
    old_hash = lookup_stored_hash(changed_file)
    new_hash = xxh3(read(changed_file))
    if new_hash == old_hash:
        return  # 跳过
    delete_nodes_and_edges_of(changed_file)
    reparse_and_reresolve(changed_file)     # 仅对该文件重跑 Phase 2-4
    recompute_affected_communities(changed_file)
    update_stored_hash(changed_file, new_hash)
```

## 3. 关键公式与数据结构

### 3.1 数学公式

**Louvain 模块度增益 (社区发现，Phase 6)：**

`ΔQ = w_in − γ · k_i · Σ_tot / (2m)`

- `w_in`：节点 i 加入目标社区后新增的社区内部边权重
- `k_i`：节点 i 的加权度（进出边权重之和）
- `Σ_tot`：目标社区所有节点的总度数
- `m`：图中全部边权重之和
- `γ = 1.0`：分辨率参数，控制社区粒度（越大越倾向于产生更多更小的社区）

物理意义：节点是否应该加入某社区，取决于"加入后带来的社区内部连接增益"是否超过"按随机图零模型预期应产生的连接数"。该公式用于 Phase 1 的局部贪心移动阶段，每个节点选择使 ΔQ 最大的相邻社区加入。算法只作用于 `CALLS`、`HTTP_CALLS`、`ASYNC_CALLS` 三类边，收敛通常 3–5 轮迭代，收敛后对内部密度 <1% 的社区做拆分精炼。

**六策略消解的置信度分级**（非严格数学公式，但是决策依据）：

`confidence ∈ {0.95, 0.90, 0.85, 0.75(或0.60), 0.55, 0.30~0.40}`

按策略优先级从高到低依次尝试，一旦命中即采用该置信度，不再降级尝试。

### 3.2 节点（Node）核心字段

| 字段 | 说明 |
|---|---|
| `id` | SQLite row id（Flush 阶段由临时 id 重映射而来） |
| `qualified_name` | 全限定名，跨文件消歧的主键 |
| `simple_name` | 简单名，用于反向索引（策略4/5的输入） |
| `label / kind` | 节点类型：Project / Package / Folder / File / Module / Function / Method / Class / Interface / Enum / Type / Route |
| `signature` | 函数/方法签名（含返回类型、形参） |
| `receiver` | 方法接收者类型（Go 场景重要） |
| `decorators` | 装饰器/注解标签 |
| `is_exported` | 是否对外可见 |
| `complexity` | 复杂度估计（用于枢纽/热点分析） |
| `file_path / span` | 源码位置，供 `get_code_snippet` 检索 |
| `content_hash` | XXH3 哈希，驱动增量重建 |
| `community_id` | Louvain 分配的社区归属（Phase 6 产出） |

### 3.3 边（Edge）核心字段

| 字段 | 说明 |
|---|---|
| `type` | CALLS / HTTP_CALLS / ASYNC_CALLS / IMPORTS / CONTAINS_* / DEFINES / DEFINES_METHOD / IMPLEMENTS / HANDLES / INHERITS / DECORATES / USES_TYPE / USAGE / THROWS / READS / WRITES / CONFIGURES / TESTS / FILE_CHANGES_WITH / MEMBER_OF |
| `src / dst` | 源节点、目标节点的 qualified_name 或 id |
| `confidence` | 0.0–1.0，标注该边的消解可信度（仅 CALLS 类边有意义；类型解析产出的边固定为高置信度） |
| `resolution_strategy` | 记录该边由哪一级策略产出（import_map / same_module / unique_name / suffix / fuzzy / type_resolved），便于审计和调优 |
| `weight` | 用于 Louvain 计算的加权度 |

## 4. 与 RepoScope 当前"符号名精确匹配"方法的差异对比

> 说明：因未能公开访问到 RepoScope 仓库内容，以下对比基于题目给定假设（现有方案 = 纯符号名精确匹配）进行推演，具体差异幅度需结合实际代码验证。

### 4.1 实体解析：如何解决同名冲突？

- **现状（精确匹配）**：对调用点的裸名做全局符号表查找，遇到同名符号（如多个模块下都有 `Parse()` 函数）时，要么随机/首个命中，要么直接产生歧义无法建边，容易产生大量误连接或漏连接。
- **论文方法的改进**：不是单一匹配，而是六级级联，且**明确携带置信度**。核心改进点：
  1. 优先用 import 语句还原出被调用符号所在的完整模块路径，再做全限定名精确匹配（策略1/2/3），从根本上避免"同名不同源"的误判；
  2. 只有在导入信息不可用时才退化为全局唯一名匹配（策略4），且此时会检查该候选是否在当前文件的 import 可达范围内并据此降权；
  3. 每条产生的边都携带 `confidence` 字段，下游查询/展示可以按置信度过滤，而不是把误判和正确匹配一视同仁地混入图谱。
- **对 RepoScope 的建议**：引入"文件级 import 表"作为消歧的第一优先信号，符号名精确匹配降级为兜底策略而非唯一手段；为每条 CALLS 边补充置信度字段。

### 4.2 关系构建：如何处理间接调用或动态分发？

- **现状（精确匹配）**：对方法接收者类型、指针解引用、接口的动态分发基本无能为力——`obj.Method()` 这种调用如果不知道 `obj` 的静态类型，精确匹配大概率匹配不到正确的实现，或者匹配到所有同名方法（过度连接）。
- **论文方法的改进**：为 Go/C/C++ 增加独立的"LSP 式混合类型解析"通道，在字符串级联之外单独运行：
  - 维护每文件的 `Scope`，跟踪变量绑定（声明/赋值/短变量声明/形参）；
  - 自底向上求值接收者表达式类型：标识符查找 → 字段/方法查找（含基类/内嵌类型遍历）→ 返回类型沿调用链传播 → 类型化简（引用展开、指针解引用、别名解析）；
  - C++ 额外处理命名空间解析和模板参数默认值，模板调用可"延迟"到具体实参类型确定后再回填解析。
  - 解析成功的调用**完全绕过**字符串级联，产出置信度更高的边。
- **对 RepoScope 的建议**：不需要一步到位支持全部语言的类型解析，可优先为项目主力语言（如 Go/TS/Python 中有类型标注的部分）实现一个轻量的作用域+类型传播通道，作为对符号匹配结果的"精化层"，而不是推倒重来。

### 4.3 增量更新：如何避免全量重建？

- **现状（精确匹配）**：若无内容哈希机制，通常只能靠文件 mtime 或者干脆全量重扫，代价随仓库规模线性增长，且 mtime 在 CI/git checkout 场景下不可靠（clone 后所有文件 mtime 相同）。
- **论文方法的改进**：
  - 用 XXH3（非加密哈希，吞吐量约 30 GB/s）对文件内容做指纹比对，只有内容真正变化才触发重建，避免 mtime 假阳性；
  - 重建粒度精确到"文件"：删除该文件对应的节点/边，仅重新解析该文件并重跑 Resolution/Enrichment，而不是全库重扫；
  - 受影响的 Louvain 社区做局部重计算，而非全图重新聚类；
  - 论文报告增量重建相比全量重建约有 4 倍加速（Table 8）。
- **对 RepoScope 的建议**：引入内容哈希（而非仅靠 mtime）作为变更检测的唯一真值来源；将"删除该文件旧图数据 + 局部重解析"设计为标准增量路径，为后续接入文件监听（file watcher）打基础。

## 5. 实施可行性评估

| 维度 | 评估 |
|---|---|
| **侵入性等级** | **中**。核心改动集中在图谱构建模块（消歧算法替换、增量更新逻辑），但要让"置信度""resolution_strategy""content_hash"这些新字段真正生效，需要对现有图数据模型（Node/Edge schema）和索引流水线做扩展，因此不属于"低"侵入；只要复用现有 Tree-Sitter/AST 提取层和存储引擎，不必推翻底层存储，故不到"高"侵入级别。 |
| **新增依赖** | 原则上**不需要**引入图数据库或嵌入模型——论文本身也强调用单文件 SQLite + 属性图模型即可实现全部功能，这与"零基础设施"的设计目标一致。若要复刻类型解析通道（Go/C/C++），需要额外的作用域/类型追踪代码，但不引入第三方库；若 RepoScope 现有存储不是关系型/嵌入式数据库，则需评估迁移成本。字符串相似度兜底（策略6）可以用轻量级编辑距离实现，无需引入 ML 依赖。 |
| **性能预期** | 索引耗时：新增的 import-map 构建、类型解析通道会带来**一定比例的额外解析时间**（论文未给出该子步骤单独耗时，但全量索引一个 49K 节点仓库约 6 秒、Linux 内核级别约 3 分钟，量级上仍可接受）；增量重建耗时预期下降到全量的 1/4 左右（依据论文报告的加速比）。查询准确率：论文的六策略级联相较单一精确匹配，在真实仓库上据称可覆盖约 80% 的调用（策略1-3），加上类型解析和模糊兜底，边覆盖率和准确率均有提升空间；具体提升幅度需在 RepoScope 自身代码库上做 A/B 测才能确定，不能直接照搬论文的 83%/92% 数字（那是端到端 agent 问答质量，不是单纯的边消歧准确率）。 |

## 6. 最小验证方案

以下脚本**完全独立**于 RepoScope 项目，使用模拟的"函数定义 + 调用点"数据，验证论文核心机制——**六策略级联消歧算法**——相较"朴素精确匹配"的差异。可直接运行（仅用标准库）。

```python
"""
minimal_disambiguation_poc.py
验证：六策略级联消歧 vs 朴素符号名精确匹配
运行：python minimal_disambiguation_poc.py
"""

from dataclasses import dataclass, field
from difflib import SequenceMatcher


@dataclass
class Entity:
    qualified_name: str   # e.g. "pkg_a.Parse"
    simple_name: str      # e.g. "Parse"


@dataclass
class CallSite:
    caller: str            # 发起调用的实体 qualified_name
    raw_callee: str         # 源码里写的调用文本，如 "utils.Parse" 或 "Parse"
    current_module: str     # 调用发生所在的模块 qualified_name
    import_map: dict        # {alias: module_qualified_name}


class Registry:
    def __init__(self, entities):
        self.by_qname = {e.qualified_name: e for e in entities}
        self.by_simple = {}
        for e in entities:
            self.by_simple.setdefault(e.simple_name, []).append(e)

    def exact(self, qname):
        return self.by_qname.get(qname)

    def reverse(self, simple_name):
        return self.by_simple.get(simple_name, [])

    def fuzzy(self, name, threshold=0.3):
        best, best_score = None, 0.0
        for cand_name, ents in self.by_simple.items():
            score = SequenceMatcher(None, name, cand_name).ratio()
            if score > best_score:
                best, best_score = ents[0], score
        return (best, best_score) if best_score >= threshold else (None, 0.0)


# ---------- 朴素基线：纯符号名精确匹配 ----------
def naive_resolve(call: CallSite, registry: Registry):
    simple = call.raw_callee.split(".")[-1]
    candidates = registry.reverse(simple)
    if len(candidates) == 1:
        return candidates[0], 1.0
    elif len(candidates) > 1:
        return candidates[0], 0.5  # 随意选第一个，模拟"误判风险"
    return None, 0.0


# ---------- 论文方法：六策略级联 ----------
def cascade_resolve(call: CallSite, registry: Registry):
    raw = call.raw_callee
    if "." in raw:
        prefix, suffix = raw.split(".", 1)
    else:
        prefix, suffix = None, raw

    # 策略1: import map 精确匹配
    if prefix and prefix in call.import_map:
        module_q = call.import_map[prefix]
        cand = registry.exact(f"{module_q}.{suffix}")
        if cand:
            return cand, 0.95, "import_map"

    # 策略3: 同模块前缀匹配
    cand = registry.exact(f"{call.current_module}.{raw}")
    if cand:
        return cand, 0.90, "same_module"

    # 策略4: 全局唯一简单名
    candidates = registry.reverse(suffix)
    if len(candidates) == 1:
        return candidates[0], 0.75, "unique_name"

    # 策略5: 多候选，按是否在 import 表中出现打分（简化版后缀/距离评分）
    if len(candidates) > 1:
        for c in candidates:
            owner_module = c.qualified_name.rsplit(".", 1)[0]
            if owner_module in call.import_map.values():
                return c, 0.55, "suffix_match(import-reachable)"
        return candidates[0], 0.40, "suffix_match(fallback)"

    # 策略6: 模糊匹配兜底
    cand, score = registry.fuzzy(suffix)
    if cand:
        return cand, round(0.30 + 0.10 * score, 2), "fuzzy"

    return None, 0.0, "unresolved"


def run_demo():
    # 模拟场景：两个模块下都定义了同名函数 Parse，制造"同名冲突"
    entities = [
        Entity("pkg_a.Parse", "Parse"),
        Entity("pkg_b.Parse", "Parse"),
        Entity("pkg_a.Helper", "Helper"),
    ]
    registry = Registry(entities)

    calls = [
        # case 1: 有 import，应该精确定位到 pkg_b.Parse（而非默认第一个 pkg_a.Parse）
        CallSite(caller="main.Run", raw_callee="pb.Parse",
                 current_module="main", import_map={"pb": "pkg_b"}),
        # case 2: 同模块内调用，无 import 前缀
        CallSite(caller="pkg_a.Run", raw_callee="Helper",
                 current_module="pkg_a", import_map={}),
        # case 3: 无 import 信息、且全局同名冲突（真正的歧义场景）
        CallSite(caller="unknown.Run", raw_callee="Parse",
                 current_module="unknown", import_map={}),
    ]

    print(f"{'Case':<6}{'Naive Result':<20}{'Cascade Result':<25}{'Strategy':<15}{'Confidence'}")
    for i, call in enumerate(calls, 1):
        naive_ent, naive_conf = naive_resolve(call, registry)
        cas_ent, cas_conf, strategy = cascade_resolve(call, registry)

        naive_name = naive_ent.qualified_name if naive_ent else "None"
        cas_name = cas_ent.qualified_name if cas_ent else "None"

        print(f"{i:<6}{naive_name:<20}{cas_name:<25}{strategy:<15}{cas_conf}")

    print("\n结论：case 1 中朴素匹配无法利用 import 信息，极可能误连到 pkg_a.Parse；")
    print("级联算法通过 import_map 精确命中 pkg_b.Parse，并给出高置信度 0.95。")
    print("case 3 中两种方法都无法完全消歧，但级联算法诚实地给出低置信度分数，")
    print("下游查询可据此过滤，而朴素方法会静默产出一条可能错误的高置信边。")


if __name__ == "__main__":
    run_demo()
```
